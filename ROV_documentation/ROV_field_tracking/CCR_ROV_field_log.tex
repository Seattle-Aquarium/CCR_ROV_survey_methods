%----------------------------------------------------------------------

\newlength{\rulew}\setlength{\rulew}{0.4pt}          % vertical hairline
\newlength{\datw}\setlength{\datw}{\dimexpr(\textwidth-0.34in-6\rulew)/6\relax}
\newlength{\notew}\setlength{\notew}{\dimexpr 6\datw+5\rulew\relax}
\newlength{\notewb}\setlength{\notewb}{\dimexpr\notew+\rulew\relax}

% Row struts: total row height = strut height + strut depth.
% Sized so six transect blocks fill the page down to the footer rule.
% The notes strut is nearly all depth, which pins the NOTES label to the top
% of the cell and opens the space below it for several lines of writing.
\newcommand{\datastrut}{\rule[-10.5pt]{0pt}{26pt}}    % 36.5pt of writing room
\newcommand{\notestrut}{\rule[-40pt]{0pt}{14pt}}      % 54pt

%----------------------------------------------------------------------
%  Small building blocks
%----------------------------------------------------------------------
% Empty check box, drawn in Salish
\newcommand{\cb}{\tikz[baseline=-0.55ex]{%
  \draw[Salish,line width=0.7pt,rounded corners=0.7pt]
    (0,-0.40em) rectangle (1.00em,0.60em);}}

% Field label sitting to the left of a write-on rule
\newcommand{\lbl}[1]{{\fontsize{7.4}{8.4}\selectfont\bfseries\color{Salish}%
  \MakeUppercase{#1}}\hspace{5pt}}

% One pre-dive checklist item
\newcommand{\pdi}[1]{{\fontsize{8.6}{10}\selectfont #1}}

% Stacked column heading: name over a Seafoam unit hint
\newcommand{\hd}[2]{%
  \begin{tabular}{@{}c@{}}
    {\fontsize{7.6}{8.6}\selectfont\bfseries\color{white}\MakeUppercase{#1}}\\[0.6pt]
    {\fontsize{5.6}{6.2}\selectfont\color{Seafoam}#2}
  \end{tabular}}

% One transect block: a data row above a full-width notes row.
% \cline{2-7} leaves column 1 untouched so the Salish number rail runs
% unbroken through the pair; the closing \hline then separates transects.
\newcommand{\transect}[1]{%
  \textcolor{white}{\bfseries\fontsize{9}{10}\selectfont #1}\datastrut
    & & & & & & \\
  \arrayrulecolor{Stone!25}\cline{2-7}
  & \multicolumn{6}{@{}>{\columncolor{Pumice}}p{\notew}@{}}{%
      \notestrut\hspace{5pt}%
      {\fontsize{6}{7}\selectfont\color{Stone}\MakeUppercase{notes}}} \\
  \arrayrulecolor{Stone!45}\hline
}
